#include <SPI.h>
#include <RF24.h>

#define CE_PIN  4
#define CSN_PIN 5

RF24 radio(CE_PIN, CSN_PIN);

const byte address[6] = "00001";

void setup() {
  Serial.begin(115200);

  if (!radio.begin()) {
    Serial.println("NRF24L01 NOT FOUND");
    while (1);
  }

  radio.setPALevel(RF24_PA_LOW);
  radio.setDataRate(RF24_250KBPS);
  radio.setChannel(108);

  radio.openWritingPipe(address);
  radio.stopListening();

  Serial.println("TRANSMITTER READY");
}

void loop() {
  byte signal = 1;

  bool success = radio.write(&signal, sizeof(signal));

  if (success) {
    Serial.println("Signal sent");
  } else {
    Serial.println("Send failed");
  }

  delay(1000);
}


الثاني


#include <SPI.h>
#include <RF24.h>

#define CE_PIN  4
#define CSN_PIN 5

#define OUTPUT_PIN 2

RF24 radio(CE_PIN, CSN_PIN);

const byte address[6] = "00001";

void setup() {
  Serial.begin(115200);

  pinMode(OUTPUT_PIN, OUTPUT);
  digitalWrite(OUTPUT_PIN, LOW);

  if (!radio.begin()) {
    Serial.println("NRF24L01 NOT FOUND");
    while (1);
  }

  radio.setPALevel(RF24_PA_LOW);
  radio.setDataRate(RF24_250KBPS);
  radio.setChannel(108);

  radio.openReadingPipe(1, address);
  radio.startListening();

  Serial.println("RECEIVER READY");
}

void loop() {

  if (radio.available()) {

    byte signal;

    radio.read(&signal, sizeof(signal));

    if (signal == 1) {

      Serial.println("Signal received");

      digitalWrite(OUTPUT_PIN, HIGH);

      delay(200);

      digitalWrite(OUTPUT_PIN, LOW);
    }
  }
}


التؤصيل

# ESP32 + NRF24L01 Communication Test

اختبار الاتصال بين وحدتي ESP32 باستخدام NRF24L01.

## التوصيل

| NRF24L01 | ESP32   |
| -------- | ------- |
| VCC      | 3.3V    |
| GND      | GND     |
| CE       | GPIO 4  |
| CSN      | GPIO 5  |
| SCK      | GPIO 18 |
| MISO     | GPIO 19 |
| MOSI     | GPIO 23 |

## الاختبار

وحدة الإرسال ترسل إشارة تلقائيًا كل ثانية.

وحدة الاستقبال تستقبل الإشارة وتشغل GPIO 2 لمدة 200ms ثم تفصله.

## المكتبة المطلوبة

RF24 by TMRh20

## ملاحظات

* NRF24L01 يعمل على 3.3V.
* CE = GPIO 4
* CSN = GPIO 5
* SPI:

  * SCK = GPIO 18
  * MISO = GPIO 19
  * MOSI = GPIO 23
* Output = GPIO 2
* Channel = 108
* Data rate = 250KBPS
* PA Level = LOW

هذا المشروع يستخدم كاختبار أساسي للتأكد من عمل الاتصال بين ESP32 وNRF24L01.
