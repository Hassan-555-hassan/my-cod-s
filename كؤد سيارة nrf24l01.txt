المرسل

#include <SPI.h>
#include <nRF24L01.h>
#include <RF24.h>

RF24 radio(9, 10); // CE, CSN

const byte address[6] = "CAR01";

// الأزرار
const int BTN_FORWARD  = 2;
const int BTN_BACKWARD = 3;
const int BTN_RIGHT    = 4;
const int BTN_LEFT     = 5;

// البوتنشيومتر
const int POT_PIN = A0;

enum Command {
  STOP = 0,
  FORWARD,
  BACKWARD,
  RIGHT,
  LEFT
};

struct DataPacket {
  byte command;
  byte speed;
};

DataPacket data;

void setup() {
  pinMode(BTN_FORWARD, INPUT_PULLUP);
  pinMode(BTN_BACKWARD, INPUT_PULLUP);
  pinMode(BTN_RIGHT, INPUT_PULLUP);
  pinMode(BTN_LEFT, INPUT_PULLUP);

  radio.begin();

  radio.setChannel(108);
  radio.setDataRate(RF24_250KBPS);
  radio.setPALevel(RF24_PA_LOW);

  radio.openWritingPipe(address);
  radio.stopListening();

  data.command = STOP;
  data.speed = 0;
}

void loop() {

  bool forward  = digitalRead(BTN_FORWARD) == LOW;
  bool backward = digitalRead(BTN_BACKWARD) == LOW;
  bool right    = digitalRead(BTN_RIGHT) == LOW;
  bool left     = digitalRead(BTN_LEFT) == LOW;

  // السرعة من البوتنشيومتر
  int potValue = analogRead(POT_PIN);
  byte speedValue = map(potValue, 0, 1023, 0, 255);

  // إذا لم يتم الضغط على أي زر
  if (!forward && !backward && !right && !left) {
    data.command = STOP;
    data.speed = 0;
  }

  // إذا تم الضغط على أكثر من زر في نفس الوقت
  else if ((forward + backward + right + left) > 1) {
    data.command = STOP;
    data.speed = 0;
  }

  else if (forward) {
    data.command = FORWARD;
    data.speed = speedValue;
  }

  else if (backward) {
    data.command = BACKWARD;
    data.speed = speedValue;
  }

  else if (right) {
    data.command = RIGHT;
    data.speed = speedValue;
  }

  else if (left) {
    data.command = LEFT;
    data.speed = speedValue;
  }

  // إرسال الأمر كل 50ms
  radio.write(&data, sizeof(data));

  delay(50);
}


الاستقبال

#include <SPI.h>
#include <nRF24L01.h>
#include <RF24.h>

RF24 radio(9, 10); // CE, CSN

const byte address[6] = "CAR01";

// L298N
const int ENA = 5;
const int IN1 = 7;
const int IN2 = 8;

const int ENB = 6;
const int IN3 = 2;
const int IN4 = 3;

enum Command {
  STOP = 0,
  FORWARD,
  BACKWARD,
  RIGHT,
  LEFT
};

struct DataPacket {
  byte command;
  byte speed;
};

DataPacket data;

// آخر وقت وصلت فيه إشارة صحيحة
unsigned long lastPacketTime = 0;

// إذا انقطعت الإشارة أكثر من 300ms تتوقف السيارة
const unsigned long FAILSAFE_TIME = 300;

void setup() {


  Serial.begin(9600);
  delay(500);

  Serial.println("==============================");
  Serial.println("     CAR RECEIVER READY");
  Serial.println("==============================");

  pinMode(ENA, OUTPUT);
  pinMode(IN1, OUTPUT);
  pinMode(IN2, OUTPUT);

  pinMode(ENB, OUTPUT);
  pinMode(IN3, OUTPUT);
  pinMode(IN4, OUTPUT);

  stopCar();

  Serial.println("L298N: READY");

  Serial.println("NRF24L01: INITIALIZING...");

  if (!radio.begin()) {
    Serial.println("NRF24L01: ERROR!");
    Serial.println("Check wiring and power.");
    while (1);
  }

  radio.setChannel(108);
  radio.setDataRate(RF24_250KBPS);
  radio.setPALevel(RF24_PA_LOW);

  radio.openReadingPipe(1, address);
  radio.startListening();

  Serial.println("NRF24L01: READY");
  Serial.println("Waiting for commands...");

  lastPacketTime = millis();

}

void loop() {

 if (radio.available()) {

  radio.read(&data, sizeof(data));

  lastPacketTime = millis();

  Serial.print("Command: ");
  Serial.print(data.command);

  Serial.print("  Speed: ");
  Serial.println(data.speed);

  executeCommand(data.command, data.speed);
}

  // FAILSAFE
  // إذا انقطعت الإشارة تتوقف السيارة
  if (millis() - lastPacketTime > FAILSAFE_TIME) {
    stopCar();
  }
}

void executeCommand(byte command, byte speed) {

  if (speed == 0 || command == STOP) {
    stopCar();
    return;
  }

  switch (command) {

    case FORWARD:
      forward(speed);
      break;

    case BACKWARD:
      backward(speed);
      break;

    case RIGHT:
      turnRight(speed);
      break;

    case LEFT:
      turnLeft(speed);
      break;

    default:
      stopCar();
      break;
  }
}

// ----------------------
// الحركة للأمام
// ----------------------

void forward(byte speed) {

  digitalWrite(IN1, HIGH);
  digitalWrite(IN2, LOW);

  digitalWrite(IN3, HIGH);
  digitalWrite(IN4, LOW);

  analogWrite(ENA, speed);
  analogWrite(ENB, speed);
}

// ----------------------
// الحركة للخلف
// ----------------------

void backward(byte speed) {

  digitalWrite(IN1, LOW);
  digitalWrite(IN2, HIGH);

  digitalWrite(IN3, LOW);
  digitalWrite(IN4, HIGH);

  analogWrite(ENA, speed);
  analogWrite(ENB, speed);
}

// ----------------------
// دوران يمين
// ----------------------

void turnRight(byte speed) {

  digitalWrite(IN1, HIGH);
  digitalWrite(IN2, LOW);

  digitalWrite(IN3, LOW);
  digitalWrite(IN4, HIGH);

  analogWrite(ENA, speed);
  analogWrite(ENB, speed);
}

// ----------------------
// دوران يسار
// ----------------------

void turnLeft(byte speed) {

  digitalWrite(IN1, LOW);
  digitalWrite(IN2, HIGH);

  digitalWrite(IN3, HIGH);
  digitalWrite(IN4, LOW);

  analogWrite(ENA, speed);
  analogWrite(ENB, speed);
}

// ----------------------
// توقف
// ----------------------

void stopCar() {

  analogWrite(ENA, 0);
  analogWrite(ENB, 0);

  digitalWrite(IN1, LOW);
  digitalWrite(IN2, LOW);

  digitalWrite(IN3, LOW);
  digitalWrite(IN4, LOW);
}
