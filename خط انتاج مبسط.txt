#include <BluetoothSerial.h>
BluetoothSerial BTSerial;

// --- تعريف المخارج والأزرار ---
const int OUT18 = 18;
const int BTN12 = 12;

const int OUT19 = 19;
const int BTN14 = 14;

const int OUT15 = 15;
const int BTN21 = 21;

const int OUT23 = 23;
const int BTN22 = 22;

const int OUT_13 = 13;
const int OUT_5 = 5;
const int BTN_RESET_5 = 27;

const int BTN_32 = 32;
const int OUT_4 = 4;

// --- مؤقتات المخرجات المؤقتة ---
unsigned long timer15_btn = 0;
unsigned long timer15_bt = 0;
const unsigned long duration15_btn = 3000;
const unsigned long duration15_bt  = 3000;

unsigned long timer23_btn = 0;
unsigned long timer23_bt = 0;
const unsigned long duration23_btn = 3000;
const unsigned long duration23_bt  = 5000;

// --- حالات البلوتوث ---
bool bt18 = false;
bool bt19 = false;
bool bt15_cmd = false;
bool bt23_cmd = false;

// --- حالات الأزرار للمخرجات المؤقتة ---
bool state15_btn = false;
bool state23_btn = false;

// --- الموجة المربعة للمخرج 13 ---
const unsigned long OUT13_ON = 5000;
const unsigned long OUT13_OFF = 4000;
unsigned long lastChange13 = 0;
bool state13 = false;

// --- المخرج 5 بعد 5 ثواني + ريست ---
const unsigned long OUT5_DELAY = 5000;
unsigned long startTime5;
bool state5 = false;

// --- الزر 32 والمخرج 4 ---
const int pressThreshold = 5;
int pressCount = 0;
bool lastButtonState32 = HIGH;
unsigned long lastPressTime32 = 0;
unsigned long output4StartTime = 0;
bool output4State = false;
const unsigned long pressDelay32 = 1500; // 1.5 ثانية بين الضغطات
const unsigned long output4Duration = 5000; // تشغيل 5 ثوانٍ

void setup() {
  pinMode(OUT18, OUTPUT);
  pinMode(OUT19, OUTPUT);
  pinMode(OUT15, OUTPUT);
  pinMode(OUT23, OUTPUT);
  pinMode(OUT_13, OUTPUT);
  pinMode(OUT_5, OUTPUT);
  pinMode(OUT_4, OUTPUT);

  pinMode(BTN12, INPUT_PULLUP);
  pinMode(BTN14, INPUT_PULLUP);
  pinMode(BTN21, INPUT_PULLUP);
  pinMode(BTN22, INPUT_PULLUP);
  pinMode(BTN_RESET_5, INPUT_PULLUP);
  pinMode(BTN_32, INPUT_PULLUP);

  BTSerial.begin("ESP32_Control");
  Serial.begin(115200);

  lastChange13 = millis();
  startTime5 = millis();
}

void loop() {
  unsigned long now = millis();

  // --- قراءة البلوتوث ---
  while(BTSerial.available()){
    char cmd = BTSerial.read();
    switch(cmd){
      case '0': bt18 = false; break;
      case '1': bt18 = true; break;
      case '2': bt19 = true; break;
      case '3': bt19 = false; break;
      case '4': bt15_cmd = true; timer15_bt = now; break;
      case '5': bt23_cmd = true; timer23_bt = now; break;
    }
  }

  // --- المخرج 18 ---
  bool btn18 = (digitalRead(BTN12) == LOW);
  digitalWrite(OUT18, btn18 || bt18 ? HIGH : LOW);

  // --- المخرج 19 ---
  bool btn19 = (digitalRead(BTN14) == LOW);
  digitalWrite(OUT19, btn19 || bt19 ? HIGH : LOW);

  // --- المخرج 15 ---
  bool btn15 = (digitalRead(BTN21) == LOW);
  if(btn15 && !state15_btn){
    timer15_btn = now;
    state15_btn = true;
  }
  bool out15 = false;
  if(state15_btn && now - timer15_btn < duration15_btn) out15 = true;
  else state15_btn = false;
  if(bt15_cmd && now - timer15_bt < duration15_bt) out15 = true;
  if(bt15_cmd && now - timer15_bt >= duration15_bt) bt15_cmd = false;
  digitalWrite(OUT15, out15 ? HIGH : LOW);

  // --- المخرج 23 ---
  bool btn23 = (digitalRead(BTN22) == LOW);
  if(btn23 && !state23_btn){
    timer23_btn = now;
    state23_btn = true;
  }
  bool out23 = false;
  if(state23_btn && now - timer23_btn < duration23_btn) out23 = true;
  else state23_btn = false;
  if(bt23_cmd && now - timer23_bt < duration23_bt) out23 = true;
  if(bt23_cmd && now - timer23_bt >= duration23_bt) bt23_cmd = false;
  digitalWrite(OUT23, out23 ? HIGH : LOW);

  // --- الموجة المربعة للمخرج 13 ---
  if (state13 && (now - lastChange13 >= OUT13_ON)) {
    state13 = false;
    digitalWrite(OUT_13, LOW);
    lastChange13 = now;
  } else if (!state13 && (now - lastChange13 >= OUT13_OFF)) {
    state13 = true;
    digitalWrite(OUT_13, HIGH);
    lastChange13 = now;
  }

  // --- المخرج 5 بعد 5 ثواني ---
  if (!state5 && (now - startTime5 >= OUT5_DELAY)) {
    state5 = true;
    digitalWrite(OUT_5, HIGH);
  }
  if (digitalRead(BTN_RESET_5) == LOW){
    state5 = false;
    digitalWrite(OUT_5, LOW);
    startTime5 = now;
  }

  // --- الزر 32 والمخرج 4 ---
  bool currentState32 = digitalRead(BTN_32);
  if(lastButtonState32 == HIGH && currentState32 == LOW){
    if(now - lastPressTime32 >= pressDelay32){
      pressCount++;
      lastPressTime32 = now;
      Serial.print("Press count: ");
      Serial.println(pressCount);
      if(pressCount >= pressThreshold){
        output4State = true;
        output4StartTime = now;
        pressCount = 0;
        Serial.println("Output 4 HIGH!");
      }
    }
  }
  lastButtonState32 = currentState32;

  // --- إدارة تشغيل المخرج 4 ---
  if(output4State && (now - output4StartTime >= output4Duration)){
    output4State = false;
    digitalWrite(OUT_4, LOW);
    Serial.println("Output 4 LOW!");
  }
  if(output4State) digitalWrite(OUT_4, HIGH);
}
