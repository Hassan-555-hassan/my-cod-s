#include <WiFi.h>
#include <WebServer.h>

const char* ssid = "Rami2.4G";
const char* password = "RAMI1985rami";

const int trigPin = 15;
const int echoPin = 16;
const int outputPin = 2;

WebServer server(80);

float getDistance() {
  digitalWrite(trigPin, LOW);
  delayMicroseconds(2);
  digitalWrite(trigPin, HIGH);
  delayMicroseconds(10);
  digitalWrite(trigPin, LOW);
  long duration = pulseIn(echoPin, HIGH);
  float distance = duration * 0.034 / 2;
  return (distance > 400 || distance == 0) ? 400 : distance;
}

String getHTML() {
  String html = "<!DOCTYPE html><html><head>";
  html += "<meta name='viewport' content='width=device-width, initial-scale=1' charset='UTF-8'>";
  html += "<style>body{background:#000; color:#0f0; text-align:center; font-family:monospace;}";
  html += "canvas{background:#001100; border:2px solid #0f0; border-radius:50%; box-shadow: 0 0 20px #0f0; max-width:90vw;}";
  html += ".info{margin-top:20px; font-size:20px;}</style></head><body>";
  html += "<h1>RADAR SYSTEM</h1><canvas id='radar' width='400' height='400'></canvas>";
  html += "<div class='info'>DISTANCE: <span id='dist'>0</span> cm</div>";
  
  html += "<script>const canvas=document.getElementById('radar'); const ctx=canvas.getContext('2d');";
  html += "let angle=0; let distance=400; function drawRadar(){";
  // رسم الدائرة والخطوط
  html += "ctx.fillStyle='rgba(0, 20, 0, 0.1)'; ctx.fillRect(0,0,400,400);";
  html += "ctx.strokeStyle='#0f0'; ctx.lineWidth=1; ctx.beginPath(); ctx.arc(200,200,180,0,Math.PI*2); ctx.stroke();";
  html += "ctx.beginPath(); ctx.arc(200,200,100,0,Math.PI*2); ctx.stroke();";
  // خط المسح
  html += "let x=200+Math.cos(angle)*180; let y=200+Math.sin(angle)*180;";
  html += "ctx.strokeStyle='#0f0'; ctx.lineWidth=3; ctx.beginPath(); ctx.moveTo(200,200); ctx.lineTo(x,y); ctx.stroke();";
  // نقطة الهدف عند الاقتراب (10 سم)
  html += "if(distance <= 10){ ctx.fillStyle='red'; ctx.beginPath(); ctx.arc(200,200,15,0,Math.PI*2); ctx.fill();";
  html += "ctx.shadowBlur=20; ctx.shadowColor='red'; } else { ctx.shadowBlur=0; }";
  html += "angle+=0.05; requestAnimationFrame(drawRadar);} drawRadar();";
  
  // تحديث البيانات من الـ ESP32
  html += "setInterval(function(){ fetch('/data').then(r=>r.json()).then(d=>{";
  html += "distance=d.distance; document.getElementById('dist').innerHTML=distance; });}, 200);</script>";
  html += "</body></html>";
  return html;
}

void setup() {
  Serial.begin(115200);
  pinMode(trigPin, OUTPUT);
  pinMode(echoPin, INPUT);
  pinMode(outputPin, OUTPUT);

  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) { delay(500); Serial.print("."); }
  
  Serial.println("\nConnected!");
  Serial.println(WiFi.localIP());

  server.on("/", []() { server.send(200, "text/html", getHTML()); });
  server.on("/data", []() {
    float d = getDistance();
    if (d <= 10 && d > 0) digitalWrite(outputPin, HIGH);
    else digitalWrite(outputPin, LOW);
    String json = "{\"distance\":" + String(d) + "}";
    server.send(200, "application/json", json);
  });
  server.begin();
}

void loop() {
  server.handleClient();
}
